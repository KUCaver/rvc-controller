cmake_minimum_required(VERSION 3.20)
project(rvc_project VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
file(GLOB_RECURSE RVC_CORE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c")
add_library(rvc_core STATIC ${RVC_CORE_SOURCES})
target_include_directories(rvc_core PUBLIC include PRIVATE src)
add_library(rvc_mock STATIC adapters/mock/mock_device.c)
target_include_directories(rvc_mock PUBLIC adapters/mock include)
add_executable(rvc_demo app/main.c)
target_link_libraries(rvc_demo PRIVATE rvc_core rvc_mock)
# The standalone program uses only the combined C/header pair.
add_executable(rvc_single_demo single_file/controller.c)
foreach(target rvc_core rvc_mock rvc_demo rvc_single_demo)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  enable_language(CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(RVC_GTEST_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/googletest-1.15.2"
      CACHE PATH "Local GoogleTest source directory")
  if(NOT EXISTS "${RVC_GTEST_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "GoogleTest is missing. Read third_party/README.md or build with -DBUILD_TESTING=OFF.")
  endif()
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  add_subdirectory("${RVC_GTEST_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
  add_executable(rvc_tests tests/rvc_tests.cpp)
  target_include_directories(rvc_tests PRIVATE src)
  target_link_libraries(rvc_tests PRIVATE rvc_core rvc_mock GTest::gtest_main)
  if(MSVC)
    target_compile_options(rvc_tests PRIVATE /W4 /WX)
  else()
    target_compile_options(rvc_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME rvc.behavior COMMAND rvc_tests --gtest_output=xml:gtest.xml)

  # Compile the very same C implementation without its CLI for the same tests.
  add_library(rvc_single_core STATIC single_file/controller.c)
  target_compile_definitions(rvc_single_core PRIVATE RVC_NO_MAIN)
  target_include_directories(rvc_single_core PUBLIC single_file)
  add_executable(rvc_single_tests tests/rvc_tests.cpp)
  target_compile_definitions(rvc_single_tests PRIVATE RVC_SINGLE_FILE)
  target_link_libraries(rvc_single_tests PRIVATE rvc_single_core GTest::gtest_main)
  foreach(target rvc_single_core rvc_single_tests)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endforeach()
  add_test(NAME rvc.single_file.behavior COMMAND rvc_single_tests --gtest_output=xml:gtest_single.xml)
endif()
